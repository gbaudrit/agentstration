const { chromium, expect } = require('/workspace/agentstration-1233/automation/playwright/node_modules/@playwright/test');
const fs=require('fs/promises');
let browser;
(async()=>{
 browser=await chromium.launch({executablePath:'/usr/bin/chromium',args:['--no-sandbox']});
 const evidence='/workspace/agentstration-1233/docs/ux/evidence/1233';await fs.mkdir(evidence,{recursive:true});const results=[];
 for(const [culture,theme,mobile] of [['en-US','dark',false],['fr-FR','light',false],['en-US','light',true],['fr-FR','dark',true]]){
  const french=culture==='fr-FR';
  const ctx=await browser.newContext({viewport:mobile?{width:390,height:844}:{width:1440,height:1000},hasTouch:mobile,isMobile:mobile,locale:culture});

  const root=page.locator('.resource-navigation');await expect(root).toHaveAttribute('data-navigation-ready','true');
  await expect(page.locator('#selected')).toHaveText('definition/tools');await expect(page).toHaveURL(/#definition\/tools$/);
  await expect(page).toHaveURL(url=>![...url.searchParams.keys()].some(key=>['view','tab'].includes(key.toLowerCase())));

  const activate=async path=>{
   if(mobile){await page.locator('.resource-navigation-trigger').tap();await expect(page.locator('dialog')).toBeVisible();await page.locator(`dialog a[data-resource-section="${path}"]`).tap();await expect(page.locator('dialog')).toHaveCount(0);}
   else{const a=page.locator(`.resource-navigation-rail a[data-resource-section="${path}"]`);await a.focus();await a.press('Enter');}
   await expect(page.locator('#selected')).toHaveText(path);
  };

  await activate('yaml');await page.getByRole('textbox',{name:french?'Brouillon YAML':'YAML draft',exact:true}).fill('invalid: [ YAML draft');

  await page.goForward();await expect(page.locator('#selected')).toHaveText('yaml');await expect(page).toHaveURL(/#yaml$/);await expect(page.getByRole('textbox',{name:french?'Brouillon YAML':'YAML draft',exact:true})).toHaveValue('invalid: [ YAML draft');
  const tab=await ctx.newPage();await tab.goto(page.url());await expect(tab.locator('.resource-navigation')).toHaveAttribute('data-navigation-ready','true');await expect(tab.locator('#selected')).toHaveText('yaml');await tab.close();
  await page.evaluate(()=>Blazor.navigateTo(location.href.split('#')[0]+'#definition/tools'));await expect(page.locator('#selected')).toHaveText('definition/tools');
  await expect(page.locator('#initializations')).toHaveText('1');
  expect(await page.locator('.fixture-root').evaluate(e=>getComputedStyle(e).getPropertyValue('--color-surface').trim())).toBe(theme==='dark'?'#111c2b':'#fff');
  await page.screenshot({path:`${evidence}/${culture}-${theme}-component.png`,fullPage:true});
  if(mobile){
   const trigger=page.locator('.resource-navigation-trigger');await trigger.tap();const dialog=page.locator('dialog');await expect(dialog).toBeVisible();
   await dialog.getByRole('button',{name:french?'Fermer les sections':'Close sections',exact:true}).focus();await page.keyboard.press('Shift+Tab');expect(await page.evaluate(()=>!!document.activeElement.closest('dialog'))).toBe(true);
   await page.screenshot({path:`${evidence}/${culture}-${theme}-drawer.png`,fullPage:true});
   await page.keyboard.press('Escape');await expect(dialog).toHaveCount(0);await expect(trigger).toBeFocused();
   await trigger.tap();await expect(dialog).toBeVisible();await page.touchscreen.tap(3,200);await expect(dialog).toHaveCount(0);await expect(trigger).toBeFocused();
   await trigger.tap();await expect(dialog).toBeVisible();await page.setViewportSize({width:1440,height:1000});await expect(dialog).toHaveCount(0);
   await expect(page.locator('.resource-navigation-rail a[aria-current=page]')).toBeFocused();
   await page.setViewportSize({width:390,height:844});await expect(trigger).toBeFocused();
   await expect(page.getByRole('textbox',{name:french?'Brouillon formulaire':'Form draft',exact:true})).toHaveValue('unsaved form');
  }

 }

})().catch(async e=>{console.error(e);await browser?.close();process.exitCode=1});
